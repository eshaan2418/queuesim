cmake_minimum_required(VERSION 3.25)
project(queuesim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(QUEUESIM_SANITIZE "" CACHE STRING "address,undefined | thread | empty")
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror -Wno-sign-conversion)
if(QUEUESIM_SANITIZE)
  add_compile_options(-fsanitize=${QUEUESIM_SANITIZE} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${QUEUESIM_SANITIZE})
endif()
find_package(ZLIB REQUIRED)
add_library(queuesim src/gz_reader.cpp)
target_include_directories(queuesim PUBLIC include)
target_link_libraries(queuesim PUBLIC ZLIB::ZLIB)
add_executable(queuesim-cli tools/queuesim.cpp)
set_target_properties(queuesim-cli PROPERTIES OUTPUT_NAME queuesim)
target_link_libraries(queuesim-cli PRIVATE queuesim)
include(FetchContent)
FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.17.0.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE SYSTEM)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)
foreach(t gtest gtest_main gmock gmock_main)
  if(TARGET ${t})
    target_compile_options(${t} PRIVATE -w)
  endif()
endforeach()
enable_testing()
add_executable(queuesim-tests tests/test_queue.cpp tests/test_sim.cpp)
target_link_libraries(queuesim-tests PRIVATE queuesim GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(queuesim-tests DISCOVERY_TIMEOUT 60)
